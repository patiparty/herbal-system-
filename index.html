ระบบยาสมุนไพร รพ.สต.

เชื่อมต่อกับ Supabase เพื่อให้ข้อมูลซิงก์แบบ real-time ทุกคนในทีมเปิดได้จากลิงก์เดียวกัน

📋 ขั้นตอนตั้งค่า (ครั้งแรกครั้งเดียว)

  1. สมัครฟรีที่ supabase.com → กด Start your project
  2. กด New Project ตั้งชื่อ herbal-db และตั้ง Password
  3. รอ ~2 นาที แล้วไปที่ SQL Editor ในเมนูซ้าย
  4. วาง SQL ด้านล่างแล้วกด Run
  5. ไปที่ Project Settings → API แล้วคัดลอก URL และ anon key
  6. กรอกด้านล่างแล้วกดเชื่อมต่อ ✅
-- วางใน SQL Editor แล้วกด Run
CREATE TABLE IF NOT EXISTS herbal_items (
  id TEXT PRIMARY KEY,
  period TEXT NOT NULL,
  name TEXT,
  carry NUMERIC DEFAULT 0,
  recv NUMERIC DEFAULT 0,
  dispense NUMERIC DEFAULT 0,
  unit TEXT DEFAULT 'เม็ด',
  benefits TEXT,
  poster_url TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS herbal_headers (
  period TEXT PRIMARY KEY,
  carry_label TEXT DEFAULT 'ยอดยกมา',
  recv_label TEXT DEFAULT 'รับเข้า'
);

-- เปิด Realtime
ALTER PUBLICATION supabase_realtime ADD TABLE herbal_items;
ALTER PUBLICATION supabase_realtime ADD TABLE herbal_headers;

-- เปิดสิทธิ์อ่าน-เขียน (RLS off สำหรับทีมภายใน)
ALTER TABLE herbal_items DISABLE ROW LEVEL SECURITY;
ALTER TABLE herbal_headers DISABLE ROW LEVEL SECURITY;
ได้จาก Project Settings → API → Project URL
